# tests/integration/test_mock_integration.py
# Teste de integração simples contra o mock_client (http://127.0.0.1:8080/status)

import requests
import pytest

BASE = "http://127.0.0.1:8080"

def test_mock_client_status():
    r = requests.get(f"{BASE}/status", timeout=3)
    assert r.status_code == 200
    data = r.json()
    # validações básicas do payload do mock
    assert "voc" in data
    assert data["voc"] == "Paladin"
    assert "pos" in data and all(k in data["pos"] for k in ("x","y","z"))
